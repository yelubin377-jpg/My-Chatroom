size_t write_callback(void *contents, size_t size, size_t nmemb, void *userp)
{
    ((std::string *)userp)->append((char *)contents, size * nmemb);
    return size * nmemb;
}

//libcurl读回调函数
static size_t read_cb(char *buf, size_t size, size_t n, void *userp)
{
    std::string *s = static_cast<std::string *>(userp);
    size_t len = std::min(size * n, s->size());
    if (len)
        memcpy(buf, s->data(), len);
    s->erase(0, len);
    return len; // 返回 0 表示 EOF
}

bool sendEmail(const std::string &recipient, const std::string &subject, const std::string &body, const std::string &smtpServer, const std::string &username, const std::string &password)
{
    CURL *curl;
    CURLcode res = CURLE_OK;
    struct curl_slist *recipients = NULL;
    std::string readBuffer;

    curl_global_init(CURL_GLOBAL_DEFAULT);
     
    curl = curl_easy_init();
    curl_easy_setopt(curl, CURLOPT_VERBOSE, 1L);
    if (curl)
    {
        // 设置收件人
        recipients = curl_slist_append(recipients, recipient.c_str());
        curl_easy_setopt(curl, CURLOPT_MAIL_RCPT, recipients);
     
        // 设置发件人
        curl_easy_setopt(curl, CURLOPT_MAIL_FROM, username.c_str());
     
        // 设置SMTP服务器地址和端口
        curl_easy_setopt(curl, CURLOPT_URL, smtpServer.c_str());
     
        // 这里我们没有进行加密，主要是使用的465端口好像不支持
        curl_easy_setopt(curl, CURLOPT_USE_SSL, (long)CURLUSESSL_NONE);
        curl_easy_setopt(curl, CURLOPT_USERNAME, username.c_str());
        curl_easy_setopt(curl, CURLOPT_PASSWORD, password.c_str());
     
        // 设置邮件内容
        std::string message =
            "Date: Mon, 29 Nov 2021 21:55:05 +1100\r\n"
            "To: " +
            recipient + "\r\n"
                        "From: " +
            username + "\r\n"
                       "Subject: " +
            subject + "\r\n"
                      "\r\n" +
            body + "\r\n";
     
        //设置相应的回调函数和发送内容
        curl_easy_setopt(curl, CURLOPT_READFUNCTION, NULL);
        curl_easy_setopt(curl, CURLOPT_UPLOAD, 1L);
        curl_easy_setopt(curl, CURLOPT_INFILESIZE, (long)message.size());
        curl_easy_setopt(curl, CURLOPT_READDATA, &message);
        curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, write_callback);
        curl_easy_setopt(curl, CURLOPT_WRITEDATA, &readBuffer);
        curl_easy_setopt(curl, CURLOPT_READFUNCTION, read_cb);

 

        // 执行发送操作
        std::cout << "[before] perform...\n";
        res = curl_easy_perform(curl);
     
        std::cout << "[after] perform...\n";
        // 清理
        curl_slist_free_all(recipients);
        curl_easy_cleanup(curl);
    }
     
    curl_global_cleanup();
     
    if (res != CURLE_OK)
    {
        std::cerr << "Error occurred: " << curl_easy_strerror(res) << std::endl;
        return false;
    }
     
    return true;

}
